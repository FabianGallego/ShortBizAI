  async function getResponseError(response: Response) {
    const contentType =
      response.headers.get("content-type") || "";

    try {
      if (contentType.includes("application/json")) {
        const result = await response.json();

        return (
          result.error ||
          result.message ||
          "OcurriÃ³ un error."
        );
      }

      const message = await response.text();

      return message || "OcurriÃ³ un error.";

  async function handlePrepareVideo(data: VideoFormData) {
    if (isPreparingVideo || isGeneratingVideo) return;

      data.sourceType === "video" &&
      !data.videoFile
    ) {
      alert("Selecciona un video original.");
      return;

    if (
      data.sourceType !== "video" &&

      alert("Selecciona al menos una fotografÃ­a.");
      return;
    }

    if (
      data.sourceType === "multiple-images" &&
      data.images.length > 5

      alert("Puedes seleccionar un mÃ¡ximo de cinco fotografÃ­as.");
      return;
    }

    try {
      setIsPreparingVideo(true);

      setFormData(data);
      setPosterMode(false);
      setVisualVideoUrl(null);
      setVisualApproved(false);
      setNarrationScript("");

      setPreliminaryVideoUrl(null);
      setPremiumForm(INITIAL_PREMIUM_FORM);
      setBackgroundMusicFile(null);

      const body = new FormData();

      body.append("action", "prepare");
      body.append("sourceType", data.sourceType);
      body.append("duration", String(data.duration || 15));
      body.append("prompt", (data.prompt || "").trim());

      if (data.sourceType === "video") {
        if (!data.videoFile) {
          throw new Error("Selecciona un video original.");
        }

        body.append("videoFile", data.videoFile);
      } else {
        for (const image of data.images) {
          body.append("images", image);
        }
      }

      const response = await fetch("/api/video-ai", {
        method: "POST",
        body,
      });

      if (!response.ok) {
        throw new Error(await getResponseError(response));
      }

      const blob = await response.blob();

      if (!blob.size) {
        throw new Error("El servidor devolviÃ³ un video vacÃ­o.");
      }

      if (
        blob.type &&
        !blob.type.includes("video") &&
        !blob.type.includes("octet-stream")
      ) {
        throw new Error(
          "La respuesta no parece ser un archivo de video."
        );
      }

      const localUrl = URL.createObjectURL(blob);

      setPreliminaryVideoBlob(blob);
      setPreliminaryVideoUrl(localUrl);

      alert(
        "Â¡Video preliminar preparado! RevÃ­salo antes de continuar."
      );
    } catch (err) {
      console.error("ERROR PREPARANDO VIDEO:", err);

      alert(
        err instanceof Error
          ? err.message
          : "No se pudo preparar el video."
      );
    } finally {
      setIsPreparingVideo(false);
    }
  }

  function getVideoUrls(video: VideoItem): string[] {
    if (!video.urls) return [];

    try {
      if (typeof video.urls === "string") {
        const parsed = JSON.parse(video.urls);

        if (Array.isArray(parsed)) {
          return parsed.filter(
            (item): item is string =>
              typeof item === "string"
          );
        }

        if (typeof parsed === "string") {
          return [parsed];
        }

        return [];
      }

      if (Array.isArray(video.urls)) {
        return video.urls.filter(
          (item): item is string =>
            typeof item === "string"
        );
      }

      return [];
    } catch {
      return [];
    }

function detectarRol(
  index: number,
  total: number,
  prompt: string,
  offer: string,
  cta: string
): VideoSceneRole {
  const instrucciones =
    normalizarTexto(prompt);

  /*
   * Si el propietario menciona explÃ­citamente
   * la oferta al principio, hacemos que una
   * escena inicial tenga ese papel.
   */
  const ofertaPrimero =
    instrucciones.includes(
      "oferta al principio"
    ) ||
    instrucciones.includes(
      "promocion al principio"
    ) ||
    instrucciones.includes(
      "descuento al principio"
    );

  if (
    ofertaPrimero &&
    index === 0 &&
    offer
  ) {
    return "offer";
  }

  if (index === 0) {
    return "hook";
  }

  if (index === total - 1) {

  return "product";
}

function cargarImagen(src: string): Promise<HTMLImageElement> {
  return new Promise((resolve, reject) => {
    const img = new Image();

    img.onload = () => resolve(img);
    img.onerror = () => reject(new Error(`No se pudo cargar la imagen: ${src}`));

    img.src = src;
  });
}

function limitar(
  valor: number,
  minimo: number,
  maximo: number
): number {
  return Math.max(minimo, Math.min(maximo, valor));
}

function easeOutCubic(valor: number): number {
  return 1 - Math.pow(1 - valor, 3);
}

function easeInOutCubic(valor: number): number {
  return valor < 0.5
    ? 4 * valor * valor * valor
    : 1 - Math.pow(-2 * valor + 2, 3) / 2;
}

function dibujarImagenCover(
  ctx: CanvasRenderingContext2D,
  img: HTMLImageElement,
  width: number,
  height: number,
  zoom = 1,
  opacity = 1,
  offsetX = 0,
  offsetY = 0
) {
  const escalaBase = Math.max(
    width / img.width,
    height / img.height
  );

  const escala = escalaBase * zoom;

  const drawWidth = img.width * escala;
  const drawHeight = img.height * escala;

  const x = (width - drawWidth) / 2 + offsetX;
  const y = (height - drawHeight) / 2 + offsetY;

  ctx.save();

  ctx.globalAlpha = opacity;

  ctx.drawImage(
    img,
    x,
    y,
    drawWidth,
    drawHeight
  );

  ctx.restore();
}

function dibujarTextoEnvuelto(
  ctx: CanvasRenderingContext2D,
  texto: string,
  x: number,
  y: number,
  maxWidth: number,
  lineHeight: number,
  maxLines = 3,
  textAlign: CanvasTextAlign = "left"
): number {
  if (!texto) {
    return 0;
  }

  const palabras = texto.trim().split(/\s+/);
  const lineas: string[] = [];

  let lineaActual = "";

  for (const palabra of palabras) {
    const prueba = lineaActual
      ? `${lineaActual} ${palabra}`
      : palabra;

    const ancho = ctx.measureText(prueba).width;

    if (ancho <= maxWidth || !lineaActual) {
      lineaActual = prueba;
    } else {
      lineas.push(lineaActual);
      lineaActual = palabra;

      if (lineas.length >= maxLines) {
        break;
      }
    }
  }

  if (
    lineaActual &&
    lineas.length < maxLines
  ) {
    lineas.push(lineaActual);
  }

  ctx.textAlign = textAlign;
  ctx.textBaseline = "top";

  lineas.forEach((linea, index) => {
    ctx.fillText(
      linea,
      x,
      y + index * lineHeight
    );
  });

  return lineas.length * lineHeight;
}

function obtenerZoom(
  animation: VideoSceneAnimation,
  progress: number,
  emphasis: "subtle" | "normal" | "strong"
): number {
  const intensidad =
    emphasis === "strong"
      ? 1.35
      : emphasis === "subtle"
        ? 0.65
        : 1;

  /*
   * Base conocida que ya estaba funcionando.
   */
  const zoomBase =
    1.02 + progress * 0.08;

  switch (animation) {
    case "zoom-out":
      return (
        1.10 -
        progress * 0.08 * intensidad
      );

    case "zoom-in":
      return (
        1.02 +
        progress * 0.08 * intensidad
      );

    case "pan-left":
    case "pan-right":
      return zoomBase;

    default:
      return zoomBase;
  }
}

function obtenerMovimiento(
  animation: VideoSceneAnimation,
  progress: number,
  width: number,
  emphasis: "subtle" | "normal" | "strong"
) {
  const intensidad =
    emphasis === "strong"
      ? 1.4
      : emphasis === "subtle"
        ? 0.65
        : 1;

  const movimiento =
    width * 0.035 * intensidad;

  const suave =
    Math.sin(progress * Math.PI);

  if (animation === "pan-left") {
    return {
      x: -movimiento * suave,
      y: 0,
    };
  }

  if (animation === "pan-right") {
    return {
      x: movimiento * suave,
      y: 0,
    };
  }

  return {
    x: 0,
    y: suave * 5 * intensidad,
  };
}

function aplicarTransicion(
  ctx: CanvasRenderingContext2D,
  previousImg: HTMLImageElement,
  img: HTMLImageElement,
  width: number,
  height: number,
  zoom: number,
  progress: number,
  transition: VideoSceneTransition
) {
  const eased = easeInOutCubic(progress);

  if (transition === "slide-left") {
    dibujarImagenCover(
      ctx,
      previousImg,
      width,
      height,
      zoom,
      1,
      -width * eased,
      0
    );

    dibujarImagenCover(
      ctx,
      img,
      width,
      height,
      zoom,
      1,
      width * (1 - eased),
      0
    );

    return;
  }

  if (transition === "slide-right") {
    dibujarImagenCover(
      ctx,
      previousImg,
      width,
      height,
      zoom,
      1,
      width * eased,
      0
    );

    dibujarImagenCover(
      ctx,
      img,
      width,
      height,
      zoom,
      1,
      -width * (1 - eased),
      0
    );

    return;
  }

  if (
    transition === "fade" ||
    transition === "crossfade"
  ) {
    dibujarImagenCover(
      ctx,
      previousImg,
      width,
      height,
      zoom,
      1 - eased
    );

    dibujarImagenCover(
      ctx,
      img,
      width,
      height,
      zoom,
      eased
    );

    return;
  }

  dibujarImagenCover(
    ctx,
    img,
    width,
    height,
    zoom,
    1
  );
}

export async function generateVideoFromScenes(
  scenes: VideoScene[],
  options: GenerateVideoOptions = {}
): Promise<Blob> {
  if (typeof window === "undefined") {
    throw new Error(
      "El generador de video solo puede ejecutarse en el navegador."
    );
  }

  if (!scenes || scenes.length === 0) {
    throw new Error(
      "No hay escenas para generar el video."
    );
  }

  const width =
    options.width ?? 1080;

  const height =
    options.height ?? 1920;

  const fps =
    options.fps ?? 30;

  const canvas =
    document.createElement("canvas");

  canvas.width = width;
  canvas.height = height;

  const ctx =
    canvas.getContext("2d");

  if (!ctx) {
    throw new Error(
      "No se pudo crear el contexto del canvas."
    );
  }

  /*
   * CORRECCIÃ“N:
   * Captura manual para sincronizar los fotogramas
   * con el tiempo de renderizado.
   */
  const stream = canvas.captureStream(0);

  const videoTrack =
    stream.getVideoTracks()[0] as
      | CanvasCaptureMediaStreamTrack
      | undefined;

  if (
    !videoTrack ||
    typeof videoTrack.requestFrame !== "function"
  ) {
    stream.getTracks().forEach((track) => {
      track.stop();
    });

    throw new Error(
      "Este navegador no admite captura manual de fotogramas. Actualiza Chrome o Edge."
    );
  }

  const mimeTypes = [
    "video/webm;codecs=vp9",
    "video/webm;codecs=vp8",
    "video/webm",
  ];

  const mimeType =
    mimeTypes.find((type) =>
      MediaRecorder.isTypeSupported(
        type
      )
    ) ?? "video/webm";

  const recorder =
    new MediaRecorder(
      stream,
      {
        mimeType,
        videoBitsPerSecond:
          10_000_000,
      }
    );

  const chunks: Blob[] = [];

  const grabar =
    new Promise<Blob>(
      (resolve, reject) => {
        recorder.ondataavailable = (
          event
        ) => {
          if (
            event.data &&
            event.data.size > 0
          ) {
            chunks.push(
              event.data
            );
          }
        };

        recorder.onerror = () => {
          reject(
            new Error(
              "OcurriÃ³ un error durante la generaciÃ³n del video."
            )
          );
        };

        recorder.onstop = () => {
          resolve(
            new Blob(
              chunks,
              {
                type: mimeType,
              }
            )
          );
        };
      }
    );

  const images =
    await Promise.all(
      scenes.map((scene) =>
        cargarImagen(
          scene.imageUrl
        )
      )
    );

  recorder.start();

  for (
    let sceneIndex = 0;
    sceneIndex < scenes.length;
    sceneIndex++
  ) {
    const scene =
      scenes[sceneIndex];

    const img =
      images[sceneIndex];

    const previousImg =
      sceneIndex > 0
        ? images[
            sceneIndex - 1
          ]
        : null;

    const duration = Math.max(
      scene.duration ?? 2,
      0.5
    );

    const totalFrames =
      Math.max(
        1,
        Math.round(
          duration * fps
        )
      );

    const transitionFrames =
      Math.min(
        Math.round(
          fps * 0.45
        ),
        Math.floor(
          totalFrames * 0.30
        )
      );

    for (
      let frame = 0;
      frame < totalFrames;
      frame++
    ) {
      const progress =
        totalFrames > 1
          ? frame /
            (totalFrames - 1)
          : 1;

      ctx.clearRect(
        0,
        0,
        width,
        height
      );

      ctx.fillStyle =
        "#000000";

      ctx.fillRect(
        0,
        0,
        width,
        height
      );

        scene.animation ??
        "zoom-in";

      const emphasis =
        scene.emphasis ??
        "normal";

      const zoom =
        obtenerZoom(
          animation,
          progress,
          emphasis
        );

      const movimiento =
        obtenerMovimiento(
          animation,
          progress,
          width,
          emphasis
        );

      if (
        previousImg &&
        transitionFrames > 0 &&
        frame < transitionFrames
      ) {
        const transitionProgress =
          easeInOutCubic(
            frame /
              transitionFrames
          );

        aplicarTransicion(
          ctx,
          previousImg,
          img,
          width,
          height,
          zoom,
          transitionProgress,
          scene.transition ??
            "fade"
        );
      } else {
        dibujarImagenCover(
          ctx,
          img,
          width,
          height,
          zoom,
          1,
          movimiento.x,
          movimiento.y
        );
      }

      /*
       * TÃ­tulo: entrada suave, permanencia y salida completa.
       * No se dibuja ningÃºn tÃ­tulo de otra escena ni texto persistente.
       */
      const fadeInEnd = 0.12;
      const fadeOutStart = 0.62;
      const fadeOutEnd = 0.82;
      let contenidoOpacity = 0;

      if (progress < fadeInEnd) {
        contenidoOpacity = easeOutCubic(progress / fadeInEnd);
      } else if (progress < fadeOutStart) {
        contenidoOpacity = 1;
      } else if (progress < fadeOutEnd) {
        contenidoOpacity = 1 - easeInOutCubic(
          (progress - fadeOutStart) / (fadeOutEnd - fadeOutStart)
        );
      }

      if (scene.title?.trim() && contenidoOpacity > 0.01) {
        const desplazamientoTexto = (1 - contenidoOpacity) * 18;
        dibujarContenidoPremium(
          ctx,
          scene,
          width,
          height,
          contenidoOpacity,
          desplazamientoTexto
        );
      }

      /*
       * CORRECCIÃ“N:
       * solicita la captura del fotograma dibujado
       * y espera el intervalo correspondiente a los FPS.
       */
      videoTrack.requestFrame();

      await new Promise<void>((resolve) => {
        window.setTimeout(
          resolve,
          1000 / fps
        );
      });
    }
  }

  /*
   * Esperar a que termine la grabaciÃ³n,
   * comprobar que el archivo tenga contenido
   * y liberar las pistas del video.
   */
  try {
    // Dejar tiempo para que MediaRecorder procese el Ãºltimo fotograma.
    await new Promise<void>((resolve) => {
      window.setTimeout(resolve, 150);
    });

    if (recorder.state !== "inactive") {
      recorder.stop();
    }

    const blob = await grabar;

    if (blob.size === 0) {
      throw new Error(
        "La grabaciÃ³n terminÃ³, pero el video estÃ¡ vacÃ­o."
      );
    }

    return blob;
  } finally {
    stream.getTracks().forEach((track) => {
      track.stop();
    });
  }
}

        ],

/**

function obtenerRolPorIndice(

): VideoSceneRole {
  if (index === 0) {

  if (index === totalScenes - 1) {

/**

 * de todas las escenas para que el video
 * tenga exactamente la duraciÃ³n indicada.
 */
export function normalizeSceneDurations(
  scenes: VideoScene[],
  totalDuration: number
): VideoScene[] {
  if (
    scenes.length === 0 ||
    !Number.isFinite(totalDuration) ||
    totalDuration <= 0
  ) {
    return scenes;
  }

  const durationPerScene =
    totalDuration / scenes.length;

  return scenes.map((scene) => ({
    ...scene,
    duration: durationPerScene,
  }));
}

async function probeMedia(filePath: string): Promise<{ duration: number; hasAudio: boolean }> {
  return new Promise((resolve, reject) => {

    const child = spawn(
      ffprobePath,
      [
        "-v", "error",
        "-show_entries", "format=duration:stream=codec_type",
        "-of", "json",
        filePath,
      ],
      { windowsHide: true, stdio: ["ignore", "pipe", "pipe"] }
    );

    let stdout = "";
    let stderr = "";
    child.stdout.on("data", (chunk: Buffer) => { stdout += chunk.toString(); });
    child.stderr.on("data", (chunk: Buffer) => { stderr = (stderr + chunk.toString()).slice(-4000); });
    child.once("error", reject);
    child.once("close", (code) => {
      if (code !== 0) {
        reject(new Error(`No se pudo analizar el MP4 con ffprobe: ${stderr}`));
        return;
      }
      try {
        const parsed = JSON.parse(stdout);
        const duration = Number(parsed.format?.duration);
        if (!Number.isFinite(duration) || duration <= 0) {
          throw new Error("El video aprobado no tiene una duraciÃ³n vÃ¡lida.");
        }
        resolve({
          duration,
          hasAudio: Array.isArray(parsed.streams) && parsed.streams.some((stream: { codec_type?: string }) => stream.codec_type === "audio"),
        });
      } catch (error) {
        reject(error);
      }
    });
  });
}

      reject(

        new Error(

          `No se pudo ejecutar FFmpeg. InstÃ¡lalo en el servidor o configura FFMPEG_PATH. ${error.message}`

        )

      );

    });



    child.once("close", (code) => {

      if (code === 0) {

        resolve();

      } else {

        reject(

          new Error(

            `FFmpeg terminÃ³ con cÃ³digo ${code}: ${stderr}`

          )

        );

      }

    });

  });

}



/* =========================================================

   DESCARGA DE RECURSOS

   Los reintentos solo se aplican a descargas HTTP.

========================================================= */



async function downloadFile(

  url: string,

  destination: string

): Promise<void> {

  let lastError: unknown;



  for (let attempt = 1; attempt <= 3; attempt++) {

    try {

      const response = await fetch(url, {

        signal: AbortSignal.timeout(90000),

      });



      if (!response.ok) {

        throw new Error(

          `No se pudo descargar un recurso (${response.status}).`

        );

      }



      const bytes = Buffer.from(

        await response.arrayBuffer()

      );



      if (bytes.length === 0) {

        throw new Error("El archivo descargado estÃ¡ vacÃ­o.");

      }



      await writeFile(destination, bytes);

      return;

    } catch (error) {

      lastError = error;



      if (attempt < 3) {

        console.warn(

          `Reintento de descarga ${attempt}/3`,

          error

        );

      }

    }

  }



  throw lastError instanceof Error

    ? lastError

    : new Error("No se pudo descargar el recurso.");

}

/* =========================================================

   ETAPA 1: PREPARACIÃ“N LOCAL DE FOTOS

   No utiliza FAL ni modelos de inteligencia artificial.

========================================================= */



async function prepareImageClip(

  image: File,

  index: number,

  duration: number,

  workDir: string

): Promise<string> {

  const inputName = `photo-${index}.jpg`;

  const outputName = `photo-clip-${index}.mp4`;

  const inputPath = path.join(workDir, inputName);



  await writeFile(inputPath, Buffer.from(await image.arrayBuffer()));



  const totalFrames = Math.max(2, Math.round(duration * FPS));

  const lastFrame = totalFrames - 1;

  const progress = `on/${lastFrame}`;



  // Alterna desplazamientos suaves para dar variedad a cada escena.

  const movements = [

    { x: `(iw-iw/zoom)*${progress}`, y: `(ih-ih/zoom)*0.5`, zoom: `1+0.16*${progress}` },

    { x: `(iw-iw/zoom)*(1-${progress})`, y: `(ih-ih/zoom)*0.5`, zoom: `1+0.15*${progress}` },

    { x: `(iw-iw/zoom)*0.5`, y: `(ih-ih/zoom)*${progress}`, zoom: `1+0.17*${progress}` },

    { x: `(iw-iw/zoom)*0.5`, y: `(ih-ih/zoom)*(1-${progress})`, zoom: `1+0.15*${progress}` },

    { x: `(iw-iw/zoom)*${progress}`, y: `(ih-ih/zoom)*${progress}`, zoom: `1+0.16*${progress}` },

  ];

  const movement = movements[index % movements.length];



  const videoFilter = [

    "scale=1080:1920:force_original_aspect_ratio=increase",

    "crop=1080:1920",

    `zoompan=z='${movement.zoom}':x='${movement.x}':y='${movement.y}':d=1:s=${WIDTH}x${HEIGHT}:fps=${FPS}`,

    "eq=contrast=1.06:saturation=1.10:brightness=0.01",

    "unsharp=5:5:0.45:5:5:0.0",

    "vignette=PI/5",

    "format=yuv420p",

  ].join(",");



  await runFfmpeg([

    "-y", "-loop", "1", "-framerate", String(FPS), "-i", inputName,

    "-vf", videoFilter, "-frames:v", String(totalFrames), "-an",

    "-c:v", "libx264", "-preset", "fast", "-crf", "18",

    "-pix_fmt", "yuv420p", "-r", String(FPS), "-movflags", "+faststart",

    outputName,

  ], workDir);



  return outputName;

}



/* =========================================================

   UNIR CLIPS CON TRANSICIONES SUAVES

========================================================= */



async function joinPreparedClips(

  clipNames: string[],

  clipDurations: number[],

  workDir: string

): Promise<string> {

  const outputName = "shortbizai-preliminary.mp4";

  const outputPath = path.join(workDir, outputName);



  if (clipNames.length === 0 || clipNames.length !== clipDurations.length) {

    throw new Error("No hay clips vÃ¡lidos para construir la previsualizaciÃ³n.");

  }



  if (clipNames.length === 1) {

    await runFfmpeg([

      "-y", "-i", clipNames[0], "-an", "-c:v", "libx264",

      "-preset", "fast", "-crf", "18", "-pix_fmt", "yuv420p",

      "-movflags", "+faststart", outputName,

    ], workDir);

    return outputPath;

  }



  const transition = 13 / FPS;

  const args: string[] = ["-y"];

  for (const clipName of clipNames) args.push("-i", clipName);



  const filters: string[] = [];

  for (let i = 0; i < clipNames.length; i++) {

    filters.push(

      `[${i}:v]settb=AVTB,setpts=PTS-STARTPTS,fps=${FPS},format=yuv420p[clip${i}]`

    );

  }



  let previousLabel = "clip0";

  for (let i = 1; i < clipNames.length; i++) {

    const offset = clipDurations

      .slice(0, i)

      .reduce((sum, value) => sum + value, 0) - transition * i;

    const outputLabel = `transition${i}`;

    filters.push(

      `[${previousLabel}][clip${i}]xfade=transition=fade:duration=${transition.toFixed(4)}:offset=${offset.toFixed(4)}[${outputLabel}]`

    );

    previousLabel = outputLabel;

  }



  filters.push(

    `[${previousLabel}]fade=t=in:st=0:d=0.25,fade=t=out:st=${Math.max(0.25, clipDurations.reduce((a, b) => a + b, 0) - transition * (clipNames.length - 1) - 0.3).toFixed(3)}:d=0.3,format=yuv420p[final]`

  );



  args.push(

    "-filter_complex", filters.join(";"), "-map", "[final]", "-an",

    "-c:v", "libx264", "-preset", "fast", "-crf", "18",

    "-pix_fmt", "yuv420p", "-r", String(FPS), "-movflags", "+faststart",

    outputName,

  );



  await runFfmpeg(args, workDir);

  return outputPath;

}

async function prepareLocally(

  formData: FormData,

  workDir: string

): Promise<NextResponse> {

  const sourceType = clean(

    formData.get("sourceType")

  );



  const duration = getDuration(

    clean(formData.get("duration"), "15")

  );



  let preliminaryPath: string;



  if (

    sourceType === "single-image" ||

    sourceType === "multiple-images"

  ) {

    const images = getImages(formData);



    const validationError = validateImages(

      sourceType,

      images

    );



    if (validationError) {

      return NextResponse.json(

        { error: validationError },

        { status: 400 }

      );

    }



    const transitionFrames = images.length > 1 ? 13 : 0;

    const requestedFrames = duration * FPS;

    const framesForAllClips =

      requestedFrames + transitionFrames * (images.length - 1);

    const baseFrames = Math.floor(framesForAllClips / images.length);

    const extraFrames = framesForAllClips % images.length;



    const clips: string[] = [];

    const clipDurations: number[] = [];



    for (let index = 0; index < images.length; index++) {

      const sceneFrames = baseFrames + (index < extraFrames ? 1 : 0);

      const sceneDuration = sceneFrames / FPS;

      const clip = await prepareImageClip(

        images[index], index, sceneDuration, workDir

      );

      clips.push(clip);

      clipDurations.push(sceneDuration);

    }



    preliminaryPath = await joinPreparedClips(

      clips,

      clipDurations,

      workDir

    );

  } else if (sourceType === "video") {

    const video = formData.get("videoFile");



    if (!(video instanceof File)) {

      return NextResponse.json(

        { error: "Selecciona un video original." },

        { status: 400 }

      );

    }



    if (video.size > MAX_VIDEO_BYTES) {

      return NextResponse.json(

        { error: "El video no puede superar los 100 MB." },

        { status: 413 }

      );

    }



    preliminaryPath = await prepareOriginalVideo(

      video,

      duration,

      workDir

    );

  } else {

    return NextResponse.json(

      { error: "Tipo de material de origen no vÃ¡lido." },

      { status: 400 }

    );

    preliminaryPath

  ) {

    return NextResponse.json(

      {

        error:

          "El video preliminar supera el tamaÃ±o permitido.",

      },

      { status: 413 }

    );

  }



  /*

   * Se devuelve el MP4 directamente.

   * El navegador lo guarda como Blob para mostrarlo

   * y enviarlo despuÃ©s si el usuario lo aprueba.

   */



  return new NextResponse(

    new Uint8Array(preliminaryBytes),

    {

      status: 200,

      headers: {

        "Content-Type": "video/mp4",

        "Content-Disposition":

          'inline; filename="shortbizai-preliminary.mp4"',

        "X-ShortBizAI-Stage": "local-preparation",

        "X-ShortBizAI-AI": "disabled",

        "Cache-Control": "no-store",

      },

    }

  );

}

): Promise<string> {
  const sourceInfo = await probeMedia(inputPath);
  const duration = sourceInfo.duration;
  const segmentCount = Math.ceil(duration / WAN_MAX_SEGMENT_SECONDS);
  const prompt = buildCinematicPrompt(fields, creativeDirection);
  const negativePrompt = "low quality, blurry, flicker, jitter, unstable camera, camera shake, distorted faces, identity change, deformed hands, extra fingers, duplicated people, duplicated objects, warped architecture, melting buildings, bent straight lines, changing landmarks, scene replacement, random stock footage, text, letters, subtitles, misspelled words, logos, watermarks, collage, split screen, flashy transitions, abrupt cuts, ghosting, frame blending artifacts, plastic skin, CGI look, oversaturated colors, excessive sharpening";
  const generatedSegments: string[] = [];

  // Repartir la duraciÃ³n equitativamente evita un Ãºltimo segmento demasiado corto.
  const segmentDuration = duration / segmentCount;
  for (let index = 0; index < segmentCount; index++) {
    const start = index * segmentDuration;
    const inputSegmentPath = path.join(workDir, `wan-input-${index + 1}.mp4`);
    const outputSegmentPath = path.join(workDir, `wan-output-${index + 1}.mp4`);

    await runFfmpeg([
      "-y", "-ss", start.toFixed(3), "-i", path.basename(inputPath),
      "-t", segmentDuration.toFixed(3), "-map", "0:v:0", "-an",
      "-vf", `scale=${WIDTH}:${HEIGHT}:force_original_aspect_ratio=increase,crop=${WIDTH}:${HEIGHT},fps=16,setsar=1`,
      "-c:v", "libx264", "-preset", "medium", "-crf", "18", "-pix_fmt", "yuv420p",
      inputSegmentPath,
    ], workDir);

    const inputBytes = await readFile(inputSegmentPath);
    const falInputFile = new File([inputBytes], `shortbizai-source-segment-${index + 1}.mp4`, { type: "video/mp4" });
    const videoUrl = await fal.storage.upload(falInputFile);
    const frameCount = Math.max(17, Math.min(161, Math.round(segmentDuration * 16)));

    const result = await fal.subscribe(WAN_VIDEO_MODEL, {
      input: {
        video_url: videoUrl,
        prompt,
        negative_prompt: negativePrompt,
        strength: 0.58,
        num_frames: frameCount,
        frames_per_second: 16,
        resolution: "720p",
        aspect_ratio: "9:16",
        num_inference_steps: 32,
        acceleration: "none",
        guidance_scale: 3.5,
        guidance_scale_2: 4,
        interpolator_model: "film",
        num_interpolated_frames: 1,
        adjust_fps_for_interpolation: true,
        video_quality: "maximum",
        video_write_mode: "balanced",
        resample_fps: true,
        enable_safety_checker: true,
        enable_output_safety_checker: true,
        enable_prompt_expansion: false,
      },
      logs: true,
      onQueueUpdate: (update: { status?: string; logs?: Array<{ message: string }> }) => {
        if (update.status === "IN_PROGRESS") {
          (update.logs || []).forEach((log) => console.log(`[WAN ${index + 1}/${segmentCount}] ${log.message}`));
        }
      },
    });

    const generatedUrl = (result.data as FalVideoResult).video?.url;
    if (!generatedUrl) {
      throw new Error(`WAN no devolviÃ³ un video para el segmento ${index + 1}.`);
    }
    await downloadFile(generatedUrl, outputSegmentPath);
    const generatedInfo = await probeMedia(outputSegmentPath);
    if (generatedInfo.duration < segmentDuration * 0.85) {
      throw new Error(`El segmento Premium ${index + 1} quedÃ³ demasiado corto (${generatedInfo.duration.toFixed(2)} s frente a ${segmentDuration.toFixed(2)} s). Se detuvo la ediciÃ³n para no entregar un video incompleto.`);
    }
    generatedSegments.push(outputSegmentPath);
  }

  if (generatedSegments.length === 1) return generatedSegments[0];

  const concatListPath = path.join(workDir, "wan-concat-list.txt");
  const concatList = generatedSegments.map((file) => `file '${path.basename(file).replace(/'/g, "'\\''")}'`).join("\n");
  await writeFile(concatListPath, concatList, "utf8");
  const joinedPath = path.join(workDir, "wan-transformed-joined.mp4");
  await runFfmpeg([
    "-y", "-f", "concat", "-safe", "0", "-i", path.basename(concatListPath),
    "-an", "-vf", `scale=${WIDTH}:${HEIGHT}:force_original_aspect_ratio=decrease,pad=${WIDTH}:${HEIGHT}:(ow-iw)/2:(oh-ih)/2:color=black,fps=${FPS},format=yuv420p`,
    "-c:v", "libx264", "-preset", "medium", "-crf", "17", "-pix_fmt", "yuv420p", joinedPath,
  ], workDir);

  const joinedInfo = await probeMedia(joinedPath);
  if (Math.abs(joinedInfo.duration - duration) > Math.max(1.0, duration * 0.08)) {
    throw new Error(`Control Premium: WAN devolviÃ³ ${joinedInfo.duration.toFixed(2)} s para un original de ${duration.toFixed(2)} s. No se publicarÃ¡ un video con una duraciÃ³n tan diferente.`);
  }
  return joinedPath;
}

/* =========================================================

   ACABADO PREMIUM

   Mezcla narraciÃ³n documental y mÃºsica sobre la producciÃ³n cinematogrÃ¡fica.

========================================================= */



async function finishPremiumVideo(
  inputPath: string,
  workDir: string,
  fields: AdFields,
  uploadedMusicPath: string | null,
  originalAudioPath: string | null = null,
  requireVoiceover = false
): Promise<string> {
  const outputPath = path.join(workDir, "shortbizai-video-final-premium.mp4");
  const sourceInfo = await probeMedia(inputPath);
  // La duraciÃ³n real del preliminar aprobado manda. Nunca se fuerza a 5 o 15 s.
  const duration = Math.max(0.5, sourceInfo.duration);
  const fadeOutStart = Math.max(0, duration - 0.35);

  const videoFilter = [
    `scale=${WIDTH}:${HEIGHT}:force_original_aspect_ratio=decrease`,
    `pad=${WIDTH}:${HEIGHT}:(ow-iw)/2:(oh-ih)/2:color=black`,
    `fps=${FPS}`,
    "setsar=1",
    "format=yuv420p",
    "fade=t=in:st=0:d=0.25",
    `fade=t=out:st=${fadeOutStart.toFixed(3)}:d=0.35`,
  ].join(",");

  let voicePath: string | null = null;
  if (process.env.FAL_KEY) {
    try {
      const voiceUrl = await generateVoiceover(fields);
      if (voiceUrl) {
        voicePath = path.join(workDir, "voiceover.mp3");
        await downloadFile(voiceUrl, voicePath);
      }
    } catch (error) {
      console.warn("No se pudo generar la narraciÃ³n.", error);
      if (requireVoiceover) throw new Error("No se pudo generar la narraciÃ³n. No se entregarÃ¡ un comercial sin la voz solicitada.");
    }
  } else if (requireVoiceover) {
    throw new Error("FAL_KEY no estÃ¡ configurada; no se puede generar la narraciÃ³n.");
  }

  let musicPath: string | null = uploadedMusicPath;
  const musicUrl = process.env.SHORTBIZAI_MUSIC_URL?.trim();
  if (!musicPath && musicUrl) {
    musicPath = path.join(workDir, "background-music.mp3");
    await downloadFile(musicUrl, musicPath);
  }

  const inputs = ["-y", "-i", inputPath];
  let nextInputIndex = 1;
  let originalAudioInputIndex: number | null = null;
  if (originalAudioPath) {
    originalAudioInputIndex = nextInputIndex++;
    inputs.push("-i", originalAudioPath);
  }
  let voiceInputIndex: number | null = null;
  if (voicePath) {
    voiceInputIndex = nextInputIndex++;
    inputs.push("-i", voicePath);
  }
  let musicInputIndex: number | null = null;
  if (musicPath) {
    musicInputIndex = nextInputIndex++;
    inputs.push("-stream_loop", "-1", "-i", musicPath);
  }

  const filters: string[] = [`[0:v:0]${videoFilter}[v]`];
  const audioLabels: string[] = [];

  if (originalAudioInputIndex !== null) {
    filters.push(`[${originalAudioInputIndex}:a:0]aresample=44100,volume=${voicePath ? 0.18 : (musicPath ? 0.22 : 1.0)},apad,atrim=duration=${duration.toFixed(3)}[source]`);
    audioLabels.push("[source]");
  }

  if (voiceInputIndex !== null) {
    filters.push(`[${voiceInputIndex}:a:0]aresample=44100,volume=1.0,apad,atrim=duration=${duration.toFixed(3)}[voice]`);
    audioLabels.push("[voice]");
  }

  if (musicInputIndex !== null) {
    filters.push(`[${musicInputIndex}:a:0]aresample=44100,volume=0.12,atrim=duration=${duration.toFixed(3)},afade=t=in:st=0:d=0.8,afade=t=out:st=${Math.max(0, duration - 1.2).toFixed(3)}:d=${Math.min(1.2, duration).toFixed(3)}[music]`);
    audioLabels.push("[music]");
  }

  const args = [...inputs];
  if (audioLabels.length > 1) {
    filters.push(`${audioLabels.join("")}amix=inputs=${audioLabels.length}:duration=longest:dropout_transition=1,alimiter=limit=0.95[a]`);
    args.push("-filter_complex", filters.join(";"), "-map", "[v]", "-map", "[a]");
  } else if (audioLabels.length === 1) {
    filters.push(`${audioLabels[0]}anull[a]`);
    args.push("-filter_complex", filters.join(";"), "-map", "[v]", "-map", "[a]");
  } else {
    args.push("-vf", videoFilter, "-map", "0:v:0", "-an");
  }

  args.push(
    "-t", duration.toFixed(3),
    "-c:v", "libx264",
    "-preset", "medium",
    "-crf", "18",
    "-pix_fmt", "yuv420p",
    "-c:a", "aac",
    "-b:a", "192k",
    "-movflags", "+faststart",
    outputPath
  );

  await runFfmpeg(args, workDir);

  const outputInfo = await probeMedia(outputPath);
  if (Math.abs(outputInfo.duration - sourceInfo.duration) > 0.15) {
    throw new Error(
      `Control de calidad: el video final dura ${outputInfo.duration.toFixed(2)} s y el preliminar ${sourceInfo.duration.toFixed(2)} s. No se entregarÃ¡ un archivo con duraciÃ³n alterada.`
    );
  }
  return outputPath;
}

/* =========================================================

   ETAPA 2: ACABADO PREMIUM DEL VIDEO APROBADO

   Recibe el preliminar aprobado por el usuario.

========================================================= */



async function generateFinalVideo(

  formData: FormData,

  workDir: string

): Promise<NextResponse> {

  const apiKey = process.env.FAL_KEY;



  if (!apiKey) {

    return NextResponse.json(

      {

        error:

          "Falta configurar FAL_KEY en el servidor.",

      },

      { status: 500 }

    );

  }



  const preparedKind = clean(formData.get("preparedKind"));
  const preparedVideo = formData.get("preparedVideo");
  let preparedPath: string;

  if (preparedKind === "poster") {
    // El afiche se convierte directamente en escenas durante la producciÃ³n final.
    preparedPath = "";
  } else {
    if (!(preparedVideo instanceof File)) {
      return NextResponse.json({ error: "No se recibiÃ³ el video preliminar aprobado." }, { status: 400 });
    }
    if (!preparedVideo.type.startsWith("video/")) {
      return NextResponse.json({ error: "El archivo preliminar no es un video vÃ¡lido." }, { status: 400 });
    }
    if (preparedVideo.size === 0 || preparedVideo.size > MAX_PREPARED_VIDEO_BYTES) {
      return NextResponse.json({ error: "El video preliminar estÃ¡ vacÃ­o o supera el tamaÃ±o permitido." }, { status: 413 });
    }
    preparedPath = path.join(workDir, "approved-preliminary.mp4");
    await writeFile(preparedPath, Buffer.from(await preparedVideo.arrayBuffer()));
  }


  const musicEntry = formData.get("backgroundMusic");
  let uploadedMusicPath: string | null = null;

  if (musicEntry instanceof File && musicEntry.size > 0) {
    if (musicEntry.size > MAX_AUDIO_BYTES) {
      return NextResponse.json(
        { error: "El audio supera el lÃ­mite de 25 MB." },
        { status: 413 }
      );
    }

    const extension = path.extname(musicEntry.name).toLowerCase();
    const allowedExtensions = [".mp3", ".wav", ".m4a", ".aac"];

    if (!allowedExtensions.includes(extension)) {
      return NextResponse.json(
        { error: "Formato de audio no permitido. Usa MP3, WAV, M4A o AAC." },
        { status: 400 }
      );
    }

    if (musicEntry.type && !musicEntry.type.startsWith("audio/")) {
      return NextResponse.json(
        { error: "El archivo seleccionado no parece ser un audio." },
        { status: 400 }
      );
    }

    uploadedMusicPath = path.join(
      workDir,
      `uploaded-background-music${extension}`
    );

    await writeFile(
      uploadedMusicPath,
      Buffer.from(await musicEntry.arrayBuffer())
    );

    // Comprueba que FFmpeg pueda leer el audio antes de consumir crÃ©ditos.
    await runFfmpeg(
      [
        "-i",
        path.basename(uploadedMusicPath),
        "-map",
        "0:a:0",
        "-t",
        "1",
        "-f",
        "null",
        "-"
      ],
      workDir
    );
  }

  if (preparedKind === "poster") {
    // Una sola acciÃ³n del usuario; el backend genera las escenas secuenciales del afiche.
    preparedPath = await generatePosterVideoPath(formData, workDir);
  }

  const creativeDirection = clean(formData.get("prompt"));
  const fields = getAdFields(formData);

  fal.config({ credentials: apiKey });



  /*

   * Guardar el preliminar en el directorio temporal.

   * El original del usuario no se modifica.

   */



  const originalInfo = await probeMedia(preparedPath);
  let originalAudioPath: string | null = null;
  if (originalInfo.hasAudio) {
    originalAudioPath = path.join(workDir, "approved-original-audio.m4a");
    await runFfmpeg(["-y", "-i", path.basename(preparedPath), "-vn", "-c:a", "aac", "-b:a", "192k", originalAudioPath], workDir);
  }

  let transformedPath: string;
  if (preparedKind === "poster") {
    // El afiche ya se convirtiÃ³ en escenas: guÃ­a, destinos y cierre de marca.
    transformedPath = preparedPath;
    console.log("SHORTBIZAI: escenas del afiche montadas en secuencia en la producciÃ³n final.");
  } else {
    console.log("SHORTBIZAI: iniciando transformaciÃ³n Premium WAN 2.2 A14B.");
    transformedPath = await generateWanTransformation(preparedPath, workDir, fields, creativeDirection);
  }

  const finalPath = await finishPremiumVideo(
    transformedPath,
    workDir,
    fields,
    uploadedMusicPath,
    originalAudioPath
  );

  const videoUrl = await uploadFinalVideo(

    finalPath

  );



  return NextResponse.json({

    success: true,

    videoUrl,

    sourceType: "approved-preliminary-video-transformed",
    aiVideoTransformation: true,

    format: "mp4",

    final: true,

  });

}

/* =========================================================
   ETAPA DE AUDIO: USA EL VIDEO VISUAL YA APROBADO

async function generateAudioForApprovedVisual(formData: FormData, workDir: string): Promise<NextResponse> {
  const apiKey = process.env.FAL_KEY;
  if (!apiKey) return NextResponse.json({ error: "Falta configurar FAL_KEY en el servidor." }, { status: 500 });

  const visualVideoUrl = clean(formData.get("visualVideoUrl"), "");
  if (!visualVideoUrl) return NextResponse.json({ error: "No se recibiÃ³ el video visual aprobado." }, { status: 400 });
  let parsedUrl: URL;
  try { parsedUrl = new URL(visualVideoUrl); } catch { return NextResponse.json({ error: "La direcciÃ³n del video aprobado no es vÃ¡lida." }, { status: 400 }); }
  if (parsedUrl.protocol !== "https:" || !(parsedUrl.hostname === "fal.media" || parsedUrl.hostname.endsWith(".fal.media"))) {
    return NextResponse.json({ error: "La direcciÃ³n del video aprobado no pertenece al almacenamiento seguro de fal.ai." }, { status: 400 });
  }

  fal.config({ credentials: apiKey });
  const visualPath = path.join(workDir, "approved-visual-only.mp4");
  await downloadFile(visualVideoUrl, visualPath);
  const visualInfo = await probeMedia(visualPath);
  if (visualInfo.duration < 1 || visualInfo.duration > 90) {
    return NextResponse.json({ error: "El video aprobado tiene una duraciÃ³n no vÃ¡lida." }, { status: 400 });
  }

  const musicEntry = formData.get("backgroundMusic");
  let uploadedMusicPath: string | null = null;
  if (musicEntry instanceof File && musicEntry.size > 0) {
    if (musicEntry.size > MAX_AUDIO_BYTES) return NextResponse.json({ error: "El audio supera el lÃ­mite de 25 MB." }, { status: 413 });
    const extension = path.extname(musicEntry.name).toLowerCase();
    if (![".mp3", ".wav", ".m4a", ".aac"].includes(extension) || (musicEntry.type && !musicEntry.type.startsWith("audio/"))) {
      return NextResponse.json({ error: "Formato de mÃºsica no vÃ¡lido. Usa MP3, WAV, M4A o AAC." }, { status: 400 });
    }
    uploadedMusicPath = path.join(workDir, `approved-music${extension}`);
    await writeFile(uploadedMusicPath, Buffer.from(await musicEntry.arrayBuffer()));
    await runFfmpeg(["-i", path.basename(uploadedMusicPath), "-map", "0:a:0", "-t", "1", "-f", "null", "-"], workDir);
  }

  const fields = getAdFields(formData);
  // Si el usuario no sube una pista propia, el Prompt 2 genera mÃºsica instrumental
  // con Stable Audio 2.5. Esta operaciÃ³n ocurre Ãºnicamente despuÃ©s de aprobar
  // el video visual y de confirmar la etapa de audio.
  if (!uploadedMusicPath && fields.audioPrompt) {
    const musicResult = await fal.subscribe("fal-ai/stable-audio-25/text-to-audio", {
      input: {
        prompt: fields.audioPrompt.slice(0, 1000),
        seconds_total: Math.max(10, Math.min(190, Math.ceil(visualInfo.duration))),
        output_format: "mp3",
        enable_safety_checker: true,
      },
      logs: true,
    });
    const generatedMusicUrl = (musicResult.data as FalAudioResult).audio?.url;
    if (!generatedMusicUrl) throw new Error("Stable Audio no devolviÃ³ la pista musical generada.");
    uploadedMusicPath = path.join(workDir, "ai-generated-background-music.mp3");
    await downloadFile(generatedMusicUrl, uploadedMusicPath);

  }

  const finalPath = await finishPremiumVideo(visualPath, workDir, fields, uploadedMusicPath, null, true);
  const videoUrl = await uploadFinalVideo(finalPath);
  return NextResponse.json({ success: true, videoUrl, stage: "complete", visualRegenerated: false });
}

/* =========================================================

   POST PRINCIPAL

   action=prepare

   action=generate-visual | generate-audio | generate-final

export async function POST(request: Request) {

  let workDir: string | null = null;



  try {

    const formData = await request.formData();



    const action = clean(

      formData.get("action")

    );



    /*

     * No se exige FAL_KEY para la preparaciÃ³n local.

     */



    if (!["prepare", "generate-visual", "generate-audio"].includes(action)) {
      return NextResponse.json(
        { error: "AcciÃ³n no vÃ¡lida. Utiliza prepare, generate-visual o generate-audio." },
        { status: 400 }
      );
    }

    workDir = await mkdtemp(path.join(tmpdir(), "shortbizai-"));

    if (action === "prepare") {
      return await prepareLocally(formData, workDir);
    }
    if (action === "generate-visual") {
      return await generateVisualOnly(formData, workDir);
    }
    return await generateAudioForApprovedVisual(formData, workDir);

  } catch (error) {

    console.error(

      "SHORTBIZAI VIDEO API ERROR:",

      error

    );



    const apiError = error as Error & {

      status?: number;

      body?: unknown;

      requestId?: string;

    };



    console.error(

      "ESTADO DEL ERROR:",

      apiError.status ?? "No disponible"

    );



    console.error(

      "DETALLE FAL:",

      JSON.stringify(

        apiError.body ?? null,

        null,

        2

      )

    );



    console.error(

      "REQUEST ID:",

      apiError.requestId ?? "No disponible"

    );



    const message =

      error instanceof Error

        ? error.message

        : "Error desconocido.";



    return NextResponse.json(

      {

        success: false,

        error:

          `No se pudo completar el proceso de video. ${message}`,

      },

      { status: 500 }

    );

  } finally {

    if (workDir) {

      await rm(

        workDir,

        {

          recursive: true,

          force: true,

        }

      ).catch(() => undefined);

    }

  }

}
